% русскоязычную вакансию (см. 06-cover-letter-templates.md) и проверка CI,
% что cover.cls набирает кириллицу: всегда собирается xelatex ровно в 1 страницу.
% Шрифты Lato 2.0 и Raleway 4.020 в OpenFonts/ содержат кириллицу; знак ₽ есть
% только в Raleway (основной текст), в Lato (имя, заголовки) его нет.
% Список стоит СНАРУЖИ \lettercontent{} (его завершающий \\ ломается на
% \end{itemize}) и обёрнут в Raleway-Medium, чтобы шрифт пунктов совпадал с текстом.

% Дата — явно по-русски: \today без настройки языка печатает английскую дату.
\currentdate{20 октября 2026 г.}
\lettercontent{Добрый день, [Имя]!}

\lettercontent{[Первый абзац: назовите должность и где нашли вакансию, одним предложением — самую сильную связь с ней и почему вы подходите. 2--3 предложения.]}

\lettercontent{[Основной абзац: самый релевантный опыт под задачи из вакансии. Дальше 3--5 конкретных пунктов:]}

{\raggedright\fontspec[Path = OpenFonts/fonts/raleway/]{Raleway-Medium}\fontsize{11pt}{13pt}\selectfont
\begin{itemize}
    \item \textbf{[Достижение 1]:} [конкретный результат, по возможности с цифрой — например, сократил время сборки на 40\,\%]
    \item \textbf{[Достижение 2]:} [навык или проект под требование вакансии]
    \item \textbf{[Достижение 3]:} [подтверждение желательного требования; зарплатные ожидания — от 250\,000~₽ на руки, если вакансия просит их указать]
\end{itemize}\par}
\vspace{6pt}

\lettercontent{[Почему именно эта компания: проверенная конкретика — продукт, заявленный приоритет, команда. Никаких общих слов.]}

\lettercontent{[Личное соответствие: сильные стороны и что вы привнесёте в команду, 2--3 предложения.]}

\lettercontent{Буду рад обсудить, чем могу быть полезен команде.}

\begin{flushright}
% Без завершающего \\ внутри \closing{}: cover.cls добавляет свой, а двойной
% перенос даёт ошибку «There's no line here to end».
\closing{С уважением,}

\signature{[ИМЯ ФАМИЛИЯ]}
\end{flushright}
